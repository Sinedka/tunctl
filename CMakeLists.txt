cmake_minimum_required(VERSION 3.20)
project(vpnctl VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
option(VPNCTL_NATIVE_SYSTEMD "Install systemd units and policy example for standalone Linux hosts" ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Curses REQUIRED)
add_library(vpnctl-core src/core.cpp src/linux.cpp)
target_include_directories(vpnctl-core PUBLIC include)
target_link_libraries(vpnctl-core PUBLIC nlohmann_json::nlohmann_json OpenSSL::Crypto)
add_executable(vpnctl src/client.cpp)
target_compile_definitions(vpnctl PRIVATE VPNCTL_IMPORT_HELPER="${CMAKE_INSTALL_FULL_LIBEXECDIR}/vpnctl-import")
target_include_directories(vpnctl PRIVATE ${CURSES_INCLUDE_DIRS})
target_link_libraries(vpnctl PRIVATE vpnctl-core ${CURSES_LIBRARIES})
add_executable(vpnctl-daemon src/daemon.cpp)
target_link_libraries(vpnctl-daemon PRIVATE vpnctl-core)
install(TARGETS vpnctl vpnctl-daemon)
install(PROGRAMS src/import.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR} RENAME vpnctl-import)
install(PROGRAMS src/dns.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR} RENAME vpnctl-dns)
install(PROGRAMS src/prepare.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR} RENAME vpnctl-prepare)
if(VPNCTL_NATIVE_SYSTEMD)
  include(packaging/native.cmake)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core-tests tests/core.cpp)
  target_link_libraries(core-tests PRIVATE vpnctl-core)
  add_test(NAME core COMMAND core-tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME profile-import COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/import_test.py)
  if(VPNCTL_NATIVE_SYSTEMD)
    add_test(NAME native-policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/native_policy.py
             ${CMAKE_CURRENT_BINARY_DIR}/vpnctl-network ${CMAKE_CURRENT_SOURCE_DIR}/packaging)
  endif()
endif()
